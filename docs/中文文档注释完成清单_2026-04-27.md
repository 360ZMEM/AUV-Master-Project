```python
def compute_los_target(pos: np.ndarray, vel: np.ndarray, path_point: np.ndarray, lookahead_dist: float) -> float:
    """计算目标航向角（视线导引）。
    
    Args:
        pos: 当前位置（北东地，米）
        vel: 当前速度（北东地，米/秒）
        path_point: 路径上最近点（北东地）
        lookahead_dist: 前视距离（米）
        
    Returns:
        float: 目标航向角（弧度，0=北，π/2=东）
        
    说明：
        - 基于视线导引律（LOS）实现路径跟踪
        - 前视距离越大，转向越平缓
        - 与 PID 航向环配合使用
    """
```

### 类级注释

```python
class AUVLocalizationNode(Node):
    """AUV 定位节点：多传感源融合与状态估计。
    
    该节点使用 ES-EKF 算法融合 IMU、DVL、深度传感器数据，
    输出滤波后的位置、速度、 姿态估计。
    
    订阅话题：
      - /imu: Imu（原始 IMU 加速度、角速度）
      - /dvl: Odometry（DVL 速度估计）
      - /depth: Float32（深度传感器）
      
    发布话题：
      - /auv/state/filtered: AUVState（滤波后的完整状态）
      
    设计原则：
      - 传感器异常时自动降级（仅 IMU）
      - 基于卡尔曼增益自适应加权
      - 线程安全（ROS2 自动保护）
    """
```
